SELECT DISTINCT
       A.ACCOUNT_NUMBER                                        AS accountNo,
       NVL(A.NEW_NAME, A.EXISTING_NAME)                        AS accountName,
       A.BRANCH_CODE                                           AS accountBranchCode,
       NULL                                                    AS accountBranchName,
       NVL(A.ACCOUNT_DOMICILE_COUNTRY_CODE,
           R.ACCT_DOMICILE_COUNTRY_CODE)                       AS accountDomicile,
       A.GFC_ID                                                AS accountGfcId,
       NVL(L.LEGAL_ENTITY_NAME, R.LEGAL_ENTITY_NAME)           AS accountLegalEntity
FROM AMT_ACCOUNT_REQUEST A
INNER JOIN AMT_REQUEST R
        ON R.REQUEST_ID = A.REQUEST_ID
LEFT JOIN AMT_PARENT_REQUEST P
       ON P.PARENT_REQUEST_ID = R.PARENT_REQUEST_ID
LEFT JOIN LEGAL_ENTITY L
       ON L.GFC_ID = A.GFC_ID
WHERE 1 = 1
  AND (:parentRequestId IS NULL OR R.PARENT_REQUEST_ID = :parentRequestId)
  AND (:requestIdsIsEmpty = 1 OR R.REQUEST_ID IN (:requestIds))
ORDER BY A.ACCOUNT_NUMBER;
